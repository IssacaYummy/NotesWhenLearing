所谓重载，就是重新赋予新的含义。函数重载是对一个已有的函数赋予新的含义，使之实现新功能

类似的，运算符也可以实现重载

运算符重载允许我们为用户定义的类型（如类或结构体）赋予运算符新的含义，这使得我们可以使用直观的语法来操作自定义对象，就像操作内置类型一样。

方法：定义一个重载运算符的函数，在需要执行被重载的运算符时，系统就自动调用该函数，以实现相应的运算

## 基本语法：

```C++
// 成员函数形式
ReturnType operatorOP(ArgumentList) {
    // 实现
}

// 全局函数形式
ReturnType operatorOP(ClassName obj, ArgumentList) {
    // 实现
}
```

其中OP是要重载的运算符号，如+， -， == 等

  

运算符重载时通过定义函数实现的 。运算符重载实质上是函数的重载

  

## 规则：

1. 保持直观性：重载的运算符应该保持其原有的语义。例如，`+` 运算符应该执行加法操作，而不是减法。
    
2. 选择成员函数还是全局函数：
    
    1. 赋值(`=`)、下标(`[]`)、函数调用(`()`)、成员访问(`->`)运算符必须作为成员函数重载
        
    2. 输入输出运算符(`<<`和`>>`)通常作为全局函数重载，并声明为类的友元
        
    3. 对称运算符(如算术运算符、相等运算符)通常作为全局函数重载
        
3. 处理自赋值：在重载赋值运算符时，要正确处理自赋值情况。
    
4. 返回类型：
    
    1. 算术运算符通常返回新对象
        
    2. 赋值运算符通常返回左侧对象的引用
        
    3. 比较运算符返回布尔值
        
5. 一致性：如果重载了 `==`，通常也应该重载 `!=`；如果重载了 `<`，通常也应该重载 `>`、`<=` 和 `>=`。
    

  

## 注意:

1. C++不允许用户自己定义新的运算符，只能对已有的运算符进行重载
    

除了下面六个运算符之外，其他运算符均允许重载：

- . (成员访问运算符)
    
- .* (成员指针运算符)
    
- :: (域运算符)
    
- Sizeof (尺寸运算符)
    
- ?: (条件运算符)
    
- typeid运算符
    

2. 重载不能改变运算符运算对象(操作数)的个数
    
3. 重载不能改变运算符的优先级别
    
4. 重载不能改变运算符的结合性
    
5. 重载的函数不能有默认的参数
    
6. 有些运算符有特殊要求，如赋值运算符必须是非静态成员函数
    
7. 重载的运算符必须和用户定义的自定义类型的对象一起使用,其参数至少应该有一个是类对象或类对象的引用(也就是说,参数不能全部都是C++的标准类型,这样约定是为了防止用户修改用于标准类型结构的运算符性质)
    

  

## 示例代码：

```C++
#include <iostream>
#include <cmath>
using namespace std;

class Complex {
private:
    double real;
    double imag;

public:
    // 构造函数
    Complex(double r = 0.0, double i = 0.0) : real(r), imag(i) {}
    
    // 获取实部和虚部
    double getReal() const { return real; }
    double getImag() const { return imag; }
    
    // 设置实部和虚部
    void setReal(double r) { real = r; }
    void setImag(double i) { imag = i; }
    
    // 重载加法运算符 (成员函数形式)
    Complex operator+(const Complex& other) const {
        return Complex(real + other.real, imag + other.imag);
    }
    
    // 重载减法运算符 (成员函数形式)
    Complex operator-(const Complex& other) const {
        return Complex(real - other.real, imag - other.imag);
    }
    
    // 重载乘法运算符 (成员函数形式)
    Complex operator*(const Complex& other) const {
        return Complex(real * other.real - imag * other.imag,
                      real * other.imag + imag * other.real);
    }
    
    // 重载除法运算符 (成员函数形式)
    Complex operator/(const Complex& other) const {
        double denominator = other.real * other.real + other.imag * other.imag;
        return Complex((real * other.real + imag * other.imag) / denominator,
                      (imag * other.real - real * other.imag) / denominator);
    }
    
    // 重载复合加法赋值运算符 (成员函数形式)
    Complex& operator+=(const Complex& other) {
        real += other.real;
        imag += other.imag;
        return *this;
    }
    
    // 重载相等运算符 (成员函数形式)
    bool operator==(const Complex& other) const {
        return (real == other.real) && (imag == other.imag);
    }
    
    // 重载不相等运算符 (成员函数形式)
    bool operator!=(const Complex& other) const {
        return !(*this == other);
    }
    
    // 重载一元负号运算符 (成员函数形式)
    Complex operator-() const {
        return Complex(-real, -imag);
    }
    
    // 重载前缀递增运算符 (成员函数形式)
    Complex& operator++() {
        ++real;
        return *this;
    }
    
    // 重载后缀递增运算符 (成员函数形式)
    Complex operator++(int) {
        Complex temp = *this;
        ++real;
        return temp;
    }
    
    // 重载类型转换运算符 (转换为double，表示模长)
    operator double() const {
        return sqrt(real * real + imag * imag);
    }
    
    // 重载函数调用运算符 (成员函数形式)
    Complex operator()(double r, double i) {
        real = r;
        imag = i;
        return *this;
    }
    
    // 友元函数声明，用于重载输出运算符
    friend ostream& operator<<(ostream& os, const Complex& c);
    
    // 友元函数声明，用于重载输入运算符
    friend istream& operator>>(istream& is, Complex& c);
};

// 重载输出运算符 (全局函数形式)
ostream& operator<<(ostream& os, const Complex& c) {
    os << c.real;
    if (c.imag >= 0) {
        os << " + " << c.imag << "i";
    } else {
        os << " - " << -c.imag << "i";
    }
    return os;
}

// 重载输入运算符 (全局函数形式)
istream& operator>>(istream& is, Complex& c) {
    cout << "请输入实部: ";
    is >> c.real;
    cout << "请输入虚部: ";
    is >> c.imag;
    return is;
}

// 重载加法运算符 (全局函数形式，用于处理 double + Complex 的情况)
Complex operator+(double d, const Complex& c) {
    return Complex(d + c.getReal(), c.getImag());
}

int main() {
    cout << "=== 运算符重载示例 ===" << endl;
    
    Complex c1(3.0, 4.0);
    Complex c2(1.0, 2.0);
    
    // 使用重载的运算符
    cout << "c1 = " << c1 << endl;
    cout << "c2 = " << c2 << endl;
    
    Complex sum = c1 + c2;
    cout << "c1 + c2 = " << sum << endl;
    
    Complex diff = c1 - c2;
    cout << "c1 - c2 = " << diff << endl;
    
    Complex product = c1 * c2;
    cout << "c1 * c2 = " << product << endl;
    
    Complex quotient = c1 / c2;
    cout << "c1 / c2 = " << quotient << endl;
    
    c1 += c2;
    cout << "c1 += c2 后，c1 = " << c1 << endl;
    
    cout << "c1 == c2: " << (c1 == c2) << endl;
    cout << "c1 != c2: " << (c1 != c2) << endl;
    
    Complex neg = -c1;
    cout << "-c1 = " << neg << endl;
    
    cout << "c1 的模长: " << (double)c1 << endl;
    
    // 测试函数调用运算符
    c1(5.0, 6.0);
    cout << "调用 c1(5.0, 6.0) 后，c1 = " << c1 << endl;
    
    // 测试全局函数重载的运算符
    Complex c3 = 2.5 + c1;
    cout << "2.5 + c1 = " << c3 << endl;
    
    // 测试输入运算符
    Complex c4;
    cin >> c4;
    cout << "你输入的复数是: " << c4 << endl;
    
    return 0;
} 
```

  

## 特殊运算符的重载

1. 下标运算符[ ]重载
    

```C++
class Array {
private:
    int* data;
    int size;
public:
    // 构造函数等省略...
    
    // 重载下标运算符
    int& operator[](int index) {
        if (index < 0 || index >= size) {
            throw out_of_range("Index out of range");
        }
        return data[index];
    }
    
    // const版本的下标运算符
    const int& operator[](int index) const {
        if (index < 0 || index >= size) {
            throw out_of_range("Index out of range");
        }
        return data[index];
    }
};
```

  

2. 函数调用运算符重载
    

```C++
class Multiplier {
private:
    int factor;
public:
    Multiplier(int f) : factor(f) {}
    
    // 重载函数调用运算符
    int operator()(int x) const {
        return factor * x;
    }
};

// 使用
Multiplier times2(2);
cout << times2(5) << endl; // 输出 10
```

3. 类型转换运算符重载
    

```C++
class Rectangle {
private:
    double width, height;
public:
    Rectangle(double w, double h) : width(w), height(h) {}
    
    // 转换为double，表示面积
    operator double() const {
        return width * height;
    }
};

// 使用
Rectangle rect(3.0, 4.0);
double area = rect; // 调用类型转换运算符，area = 12.0
```
