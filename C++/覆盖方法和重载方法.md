# 覆盖(overriding)

覆盖可以在子类中修改基类里提供的某个通用的方法的实现

我们需要做的是在类里重新声明这个方法，然后再改写一下他的实现代码（就像他是一个新增的方法一样）

# 重载方法（函数重载）

函数重载允许我们使用相同的函数名来执行相似但参数不同的操作

## 实现原理

C++通过名称修饰(name mangling)技术来实现函数重载

编译器会根据函数名和参数列表生成唯一的内部名称，这样在链接时就能区分不同的重载函数

示例代码

```C++
#include <iostream>
#include <string>
#include <cmath>
using namespace std;

// 1. 参数类型不同的重载
void print(int i) {
    cout << "整数: " << i << endl;
}

void print(double f) {
    cout << "浮点数: " << f << endl;
}

void print(const string& s) {
    cout << "字符串: " << s << endl;
}

// 2. 参数个数不同的重载
void display() {
    cout << "无参数 display()" << endl;
}

void display(int a) {
    cout << "一个参数: " << a << endl;
}

void display(int a, int b) {
    cout << "两个参数: " << a << ", " << b << endl;
}

// 3. 参数顺序不同的重载
void show(int a, double b) {
    cout << "int-double: " << a << ", " << b << endl;
}

void show(double a, int b) {
    cout << "double-int: " << a << ", " << b << endl;
}

// 4. 返回值类型不同不能作为重载依据
// int calculate() { return 0; }    // 错误：与下面的函数冲突
// double calculate() { return 0.0; } // 错误：仅返回值类型不同

// 5. 带有默认参数的重载
void log(const string& message, int level = 1) {
    cout << "日志[" << level << "]: " << message << endl;
}

void log(const string& message) {
    cout << "日志[默认]: " << message << endl;
}

// 6. 类中的成员函数重载
class Calculator {
public:
    int add(int a, int b) {
        return a + b;
    }
    
    double add(double a, double b) {
        return a + b;
    }
    
    int add(int a, int b, int c) {
        return a + b + c;
    }
};

// 7. 函数模板重载
template<typename T>
T maximum(T a, T b) {
    return (a > b) ? a : b;
}

template<typename T>
T maximum(T a, T b, T c) {
    return maximum(maximum(a, b), c);
}

int main() {
    cout << "=== 函数重载示例 ===" << endl;
    
    // 测试参数类型不同的重载
    cout << "\n1. 参数类型不同的重载:" << endl;
    print(42);
    print(3.14159);
    print("Hello, World!");
    
    // 测试参数个数不同的重载
    cout << "\n2. 参数个数不同的重载:" << endl;
    display();
    display(10);
    display(20, 30);
    
    // 测试参数顺序不同的重载
    cout << "\n3. 参数顺序不同的重载:" << endl;
    show(5, 2.5);
    show(3.7, 2);
    
    // 测试带有默认参数的重载
    cout << "\n4. 带有默认参数的重载:" << endl;
    log("一般信息");
    log("重要信息", 3);
    
    // 测试类中的成员函数重载
    cout << "\n5. 类中的成员函数重载:" << endl;
    Calculator calc;
    cout << "calc.add(2, 3) = " << calc.add(2, 3) << endl;
    cout << "calc.add(2.5, 3.7) = " << calc.add(2.5, 3.7) << endl;
    cout << "calc.add(1, 2, 3) = " << calc.add(1, 2, 3) << endl;
    
    // 测试函数模板重载
    cout << "\n6. 函数模板重载:" << endl;
    cout << "maximum(10, 20) = " << maximum(10, 20) << endl;
    cout << "maximum(2.5, 1.3) = " << maximum(2.5, 1.3) << endl;
    cout << "maximum(10, 20, 15) = " << maximum(10, 20, 15) << endl;
    
    return 0;
}
```

## 函数重载的规则

- 重载函数必须具有不同的参数列表（参数类型、个数或顺序）
    
- 不能**仅基于**返回类型的不同来重载函数
    
- 成员函数的重载也可以基于const限定符
    
- 带有默认参数的函数可能会与其他重载函数发生冲突
    
- 无法对基类继承来的方法进行重载
    

编写代码一定要注意区分重载和覆盖
