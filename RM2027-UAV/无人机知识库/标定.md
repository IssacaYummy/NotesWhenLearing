```yaml
target_type: checkerboard
targetCols: 9
targetRows: 7
rowSpacingMeters: 0.015
colSpacingMeters: 0.015
```

# 标定相机内参
终端1：发布相机话题
```bash
source /home/orangepi/px4_ws/env.sh

ros2 launch realsense2_camera rs_launch.py \
  camera_namespace:=camera \
  camera_name:=camera \
  enable_color:=false \
  enable_depth:=false \
  enable_infra1:=true \
  enable_infra2:=true \
  depth_module.infra_profile:=848x480x30 \
  depth_module.infra1_format:=Y8 \
  depth_module.infra2_format:=Y8 \
  enable_sync:=true \
  config_file:=/home/orangepi/px4_ws/realsense_emitter_off.yaml
```

终端2：查看相机话题是否发布，并关闭红外模块
```bash
source /home/orangepi/px4_ws/env.sh
ros2 topic list -t
ros2 param set /camera/camera depth_module.emitter_enabled 0
```

终端3：打开rqt查看相机图像（需要安装rqt_image_viewer插件）
```bash
source /home/orangepi/px4_ws/env.sh
rqt
```

终端4：录制rosbag
```bash
source /home/orangepi/px4_ws/env.sh

ros2 bag record -s sqlite3 \
  -o /home/orangepi/data/bags/stereo01 \
  /camera/camera/infra1/image_rect_raw \
  /camera/camera/infra2/image_rect_raw \
  /camera/camera/infra1/camera_info \
  /camera/camera/infra2/camera_info
```
