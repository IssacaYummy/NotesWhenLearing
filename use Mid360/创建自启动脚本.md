# 写总自启动脚本
以下为fastlio算法的自启动脚本
```shell
#!/bin/bash

USER_HOME=/home/scurm
LOG_DIR=$USER_HOME/uav_logs
mkdir -p $LOG_DIR

# ROS2 基础环境 + 工作空间（只需要source一次）
source /opt/ros/humble/setup.bash
export PATH=/opt/ros/humble/bin:$PATH

echo "[$(date)] 等待设备就绪..." >> $LOG_DIR/start.log
sleep 5

# =====================
# 1. Livox 驱动(FASTLIO)
# =====================
# source $USER_HOME/livox_ws/install/setup.bash
# ros2 launch livox_ros_driver2 msg_MID360_launch.py >> $LOG_DIR/livox.log 2>&1 &
# echo "[$(date)] Livox 驱动启动" >> $LOG_DIR/start.log
# sleep 3

# =====================
# 2. MID360 坐标系变换（Z轴旋转180°）          
# =====================
# source $USER_HOME/vision_pos_estimate_ws/install/setup.bash
# ros2 launch mid360_transform mid360_transform.launch.py >> $LOG_DIR/mid360_transform.log 2>&1 &
# echo "[$(date)] MID360 坐标系变换节点启动" >> $LOG_DIR/start.log
# sleep 2

# =====================
# 3. FAST-LIO2 SLAM
# =====================
source $USER_HOME/fastlio_ws/install/setup.bash
export LD_PRELOAD=/usr/lib/x86_64-linux-gnu/libusb-1.0.so.0
ros2 launch fast_lio mapping.launch.py >> $LOG_DIR/fast_lio.log 2>&1 &
echo "[$(date)] FAST-LIO2 启动" >> $LOG_DIR/start.log
sleep 3

# =====================
# 4. slam_to_vision 节点
# =====================
source $USER_HOME/vision_pos_estimate_ws/install/setup.bash
ros2 run slam_to_vision_cpp slam_to_vision_node >> $LOG_DIR/slam_to_vision.log 2>&1 &
echo "[$(date)] slam_to_vision 启动" >> $LOG_DIR/start.log
sleep 3

# =====================
# 5. MAVROS
# =====================
ros2 run mavros mavros_node --ros-args \
  -p fcu_url:=serial:///dev/ttyACM0:57600 >> $LOG_DIR/mavros.log 2>&1 &
echo "[$(date)] MAVROS 启动" >> $LOG_DIR/start.log

echo "[$(date)] 所有节点启动完毕" >> $LOG_DIR/start.log
wait
```
保存后赋予可执行权限
```bash
chmod +x ~/start_uav_fast_lio.sh
```

# 创建systemd服务
新建服务文件
```bash
cd /etc/systemd/system
sudo touch uav_autostart.service
code .
```
编辑服务文件
```bash
[Unit]
Description=UAV LiDAR SLAM + MAVROS Autostart
# 等待串口 USB 设备挂载完成后再启动
After=network-online.target systemd-udev-settle.service
Wants=systemd-udev-settle.service

[Service]
Type=simple
User=scurm
WorkingDirectory=/home/scurm
# 关键：给当前 session 注入 DISPLAY 和 DBUS，否则 ROS2 节点可能拿不到 GUI
Environment="HOME=/home/scurm"
Environment="ROS_DOMAIN_ID=0"
ExecStart=/bin/bash /home/scurm/start_uav_fast_lio.sh
# 如果进程崩溃，5秒后自动重启
Restart=on-failure
RestartSec=5
# 给足够时间等待驱动加载（特别是 LD_PRELOAD 那个 USB 库）
TimeoutStartSec=60

[Install]
WantedBy=multi-user.target
```
保存后执行
```bash
sudo systemctl daemon-reload
sudo systemctl enable uav_autostart.service
# 先不重启，先手动测试一次
sudo systemctl start uav_autostart.service
```
使用以下命令查看状态和日志
```bash
sudo systemctl status uav_autostart.service
journalctl -u uav_autostart.service -e
```
